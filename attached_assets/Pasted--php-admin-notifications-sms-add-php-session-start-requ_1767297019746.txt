if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['add_notification'])) {
    $code = strtolower(str_replace(' ', '_', trim($_POST['code'])));
    $name = trim($_POST['name']);
    $recipient_type = $_POST['recipient_type'];
    $purpose = trim($_POST['purpose']);
    $sms_content = trim($_POST['sms_content']);
    $status = $_POST['status'];
    $available_shortcodes = trim($_POST['available_shortcodes']);
    
    // Check if code already exists
    $check = $pdo->prepare("SELECT id FROM notification_templates WHERE code = ?");
    $check->execute([$code]);
    
    if ($check->rowCount() > 0) {
        $error = "A notification with this code already exists!";
    } else {
        $stmt = $pdo->prepare("INSERT INTO notification_templates 
            (type, code, name, recipient_type, purpose, sms_content, status, available_shortcodes) 
            VALUES ('sms', ?, ?, ?, ?, ?, ?, ?)");
        
        if ($stmt->execute([$code, $name, $recipient_type, $purpose, $sms_content, $status, $available_shortcodes])) {
            $_SESSION['success_message'] = "SMS notification created successfully!";
            header('Location: notifications-sms.php');
            exit;
        } else {
            $error = "Failed to create notification.";
        }
    }
}

<script>
$(document).ready(function() {
    // Character counter
    function updateCharacterCount() {
        var content = $('#sms_content').val();
        var length = content.length;
        var parts = Math.ceil(length / 160) || 1;
        var remaining = (parts * 160) - length;
        var percentage = Math.min((length / 160) * 100, 100);
        
        $('#char-count').text(length + ' characters');
        $('#current-length').text(length);
        $('#sms-parts').text(parts + ' SMS').removeClass('badge-success badge-warning badge-danger')
            .addClass(parts === 1 ? 'badge-success' : (parts === 2 ? 'badge-warning' : 'badge-danger'));
        $('#sms-count').text(parts + ' SMS Message' + (parts > 1 ? 's' : ''));
        $('#remaining-chars').text(remaining);
        $('#char-progress').css('width', percentage + '%');
        $('#sms-preview').text(content || 'Your SMS message will appear here...');
    }
    
    updateCharacterCount();
    $('#sms_content').on('input', updateCharacterCount);
    
    // Insert shortcode
    $('.insert-shortcode').click(function() {
        var shortcode = $(this).data('value');
        var textarea = $('#sms_content');
        var cursorPos = textarea[0].selectionStart;
        var content = textarea.val();
        var newContent = content.substring(0, cursorPos) + shortcode + content.substring(cursorPos);
        textarea.val(newContent);
        updateCharacterCount();
        
        $(this).text('Added!').removeClass('btn-outline-success').addClass('btn-success');
        setTimeout(() => {
            $(this).text(shortcode).removeClass('btn-success').addClass('btn-outline-success');
        }, 800);
    });
    
    // Auto-generate code from name
    $('#name').on('blur', function() {
        if ($('#code').val() === '') {
            var code = 'sms_' + $(this).val().toLowerCase()
                .replace(/[^a-z0-9\s]/g, '')
                .replace(/\s+/g, '_');
            $('#code').val(code);
        }
    });
});
</script>
