<script>
$(document).ready(function() {
    // Character counter and SMS calculator
    function updateCharacterCount() {
        var content = $('#sms_content').val();
        var length = content.length;
        var parts = Math.ceil(length / 160);
        var remaining = (parts * 160) - length;
        var percentage = (length / 160) * 100;
        
        // Update displays
        $('#char-count').text(length + ' characters');
        $('#current-length').text(length);
        $('#sms-parts').text(parts + ' SMS').removeClass('badge-success badge-warning badge-danger')
            .addClass(parts === 1 ? 'badge-success' : (parts === 2 ? 'badge-warning' : 'badge-danger'));
        $('#sms-count').text(parts + ' SMS Message' + (parts > 1 ? 's' : ''));
        $('#remaining-chars').text(remaining);
        
        // Update progress bar
        $('#char-progress').css('width', Math.min(percentage, 100) + '%')
            .removeClass('bg-success bg-warning bg-danger')
            .addClass(length <= 160 ? 'bg-success' : (length <= 320 ? 'bg-warning' : 'bg-danger'));
        
        // Update preview
        $('#sms-preview').text(content || 'Your SMS message preview will appear here...');
    }
    
    // Initial update
    updateCharacterCount();
    
    // Update on input
    $('#sms_content').on('input', updateCharacterCount);
    
    // Insert shortcode
    $('.shortcode-btn').click(function() {
        var shortcode = $(this).data('shortcode');
        var textarea = $('#sms_content');
        var cursorPos = textarea[0].selectionStart;
        var content = textarea.val();
        
        // Insert at cursor position
        var newContent = content.substring(0, cursorPos) + shortcode + content.substring(cursorPos);
        textarea.val(newContent);
        
        // Update character count
        updateCharacterCount();
        
        // Show copied feedback
        $(this).text('Inserted!').removeClass('btn-outline-success').addClass('btn-success');
        setTimeout(() => {
            $(this).text(shortcode).removeClass('btn-success').addClass('btn-outline-success');
        }, 1000);
    });
    
    // Test SMS
    $('#test-sms-btn').click(function() {
        var phone = prompt('Enter phone number for test SMS (with country code):');
        if (phone) {
            var btn = $(this);
            btn.prop('disabled', true).html('<i class="fas fa-spinner fa-spin"></i> Sending...');
            
            $.ajax({
                url: 'ajax/send-test-sms.php',
                method: 'POST',
                data: {
                    notification_id: <?php echo $notification_id; ?>,
                    phone: phone,
                    content: $('#sms_content').val()
                },
                success: function(response) {
                    alert('Test SMS sent successfully!');
                    btn.prop('disabled', false).html('<i class="fas fa-paper-plane"></i> Send Test SMS');
                },
                error: function() {
                    alert('Failed to send test SMS.');
                    btn.prop('disabled', false).html('<i class="fas fa-paper-plane"></i> Send Test SMS');
                }
            });
        }
    });
});
</script>
